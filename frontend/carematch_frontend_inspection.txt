import axios from "axios"

const api = axios.create({
  baseURL: "",
  headers: {
    "Content-Type": "application/json",
  },
})

api.interceptors.request.use(
  (config) => {
    const token = localStorage.getItem(
      "smartmedchart_token",
    )

    if (token) {
      config.headers.Authorization =
        `Bearer ${token}`
    }

    return config
  },
  (error) => {
    return Promise.reject(error)
  },
)

export default api

=== AUTH CONTEXT ===
import {
  createContext,
  useContext,
  useEffect,
  useState,
} from "react"

import {
  getCurrentUser,
  login as loginRequest,
} from "../services/auth"


const AuthContext =
  createContext(null)


export function AuthProvider({
  children,
}) {
  const [user, setUser] =
    useState(null)

  const [loading, setLoading] =
    useState(true)


  async function restoreSession() {
    const token =
      localStorage.getItem(
        "smartmedchart_token",
      )

    if (!token) {
      setUser(null)
      setLoading(false)
      return
    }


    try {
      const currentUser =
        await getCurrentUser()

      setUser(currentUser)

      localStorage.setItem(
        "smartmedchart_user",
        JSON.stringify(
          currentUser,
        ),
      )
    } catch (error) {
      console.error(
        "Session restore failed:",
        error,
      )

      localStorage.removeItem(
        "smartmedchart_token",
      )

      localStorage.removeItem(
        "smartmedchart_user",
      )

      setUser(null)
    } finally {
      setLoading(false)
    }
  }


  useEffect(() => {
    restoreSession()
  }, [])


  async function login(
    username,
    password,
  ) {
    const loginData =
      await loginRequest(
        username,
        password,
      )


    if (!loginData?.access_token) {
      throw new Error(
        "Login succeeded but no access token was returned.",
      )
    }


    localStorage.setItem(
      "smartmedchart_token",
      loginData.access_token,
    )


    /*
     * Ask backend who actually logged in.
     * This is the source of truth for the role.
     */
    const currentUser =
      await getCurrentUser()


    setUser(currentUser)


    localStorage.setItem(
      "smartmedchart_user",
      JSON.stringify(
        currentUser,
      ),
    )


    return currentUser
  }


  function logout() {
    localStorage.removeItem(
      "smartmedchart_token",
    )

    localStorage.removeItem(
      "smartmedchart_user",
    )

    setUser(null)
  }


  const value = {
    user,
    loading,
    isAuthenticated:
      Boolean(user),
    login,
    logout,
  }


  return (
    <AuthContext.Provider
      value={value}
    >
      {children}
    </AuthContext.Provider>
  )
}


export function useAuth() {
  const context =
    useContext(
      AuthContext,
    )

  if (!context) {
    throw new Error(
      "useAuth must be used inside AuthProvider",
    )
  }

  return context
}

import {

  ClipboardPlus,

  Weight,
} from "lucide-react"

import { useEffect, useState } from "react"
import { useNavigate } from "react-router-dom"

import MainLayout from "../../components/layout/MainLayout"
import api from "../../services/api"
import {
  getHospitalDoctors,
  getWardBeds,
  getWards,
} from "../../services/admin"


// =========================================================
// MAIN
// =========================================================

export default function AddPatient() {
  const navigate = useNavigate()

  // =======================================================
  // FORM
  // =======================================================

  const [form, setForm] = useState({
    first_name: "",
    last_name: "",
    date_of_birth: "",
    gender: "",
    weight_kg: "",
    doctor_id: "",
    admit_patient: false,
    ward_id: "",
    bed_id: "",
  })

  // =======================================================
  // UI STATE
  // =======================================================

  const [loading, setLoading] = useState(false)
  const [error, setError] = useState("")
  const [successPatient, setSuccessPatient] = useState(null)

  // =======================================================
  // DOCTOR STATE
  // =======================================================

  const [doctors, setDoctors] = useState([])
  const [doctorsLoading, setDoctorsLoading] = useState(false)

  // =======================================================
  // WARD / BED STATE
  // =======================================================

  const [wards, setWards] = useState([])
  const [beds, setBeds] = useState([])

  const [wardsLoading, setWardsLoading] = useState(false)
  const [bedsLoading, setBedsLoading] = useState(false)

  // =======================================================
  // LOAD DOCTORS
  // =======================================================

  async function loadDoctors() {
    setDoctorsLoading(true)
    setError("")

    try {
      const response = await getHospitalDoctors()

      const doctorList = Array.isArray(response?.doctors)
        ? response.doctors
        : []

      setDoctors(doctorList)

    } catch (requestError) {
      console.error(
        "Load doctors error:",
        requestError,
      )

      setDoctors([])

      setError(
        requestError?.response?.data?.detail ||
          "Unable to load registered doctors.",
      )

    } finally {
      setDoctorsLoading(false)
    }
  }

  // =======================================================
  // LOAD WARDS
  // =======================================================

  async function loadWards() {
    setWardsLoading(true)

    try {
      const response = await getWards()

      setWards(
        Array.isArray(response?.wards)
          ? response.wards
          : [],
      )

    } catch (requestError) {
      console.error(
        "Load wards error:",
        requestError,
      )

      setError(
        requestError?.response?.data?.detail ||
          "Unable to load hospital wards.",
      )

    } finally {
      setWardsLoading(false)
    }
  }

  // =======================================================
  // LOAD BEDS
  // =======================================================

  async function loadBeds(wardId) {
    if (!wardId) {
      setBeds([])
      return
    }

    setBedsLoading(true)

    try {
      const response = await getWardBeds(wardId)

      const availableBeds =
        Array.isArray(response?.beds)
          ? response.beds.filter(
              (bed) =>
                bed.is_active &&
                !bed.is_occupied,
            )
          : []

      setBeds(availableBeds)

    } catch (requestError) {
      console.error(
        "Load beds error:",
        requestError,
      )

      setBeds([])

      setError(
        requestError?.response?.data?.detail ||
          "Unable to load beds for this ward.",
      )

    } finally {
      setBedsLoading(false)
    }
  }

  // =======================================================
  // INITIAL LOAD
  // =======================================================

  useEffect(() => {
    loadDoctors()
    loadWards()
  }, [])

  // =======================================================
  // HANDLE INPUT
  // =======================================================

  function handleChange(event) {
    const {
      name,
      value,
      type,
      checked,
    } = event.target

    setError("")

    setForm((current) => ({
      ...current,
      [name]:
        type === "checkbox"
          ? checked
          : value,
    }))

    // -----------------------------------------------------
    // Disable admission
    // -----------------------------------------------------

    if (
      name === "admit_patient" &&
      !checked
    ) {
      setForm((current) => ({
        ...current,
        admit_patient: false,
        ward_id: "",
        bed_id: "",
      }))

      setBeds([])
    }

    // -----------------------------------------------------
    // Ward changed
    // -----------------------------------------------------

    if (name === "ward_id") {
      setForm((current) => ({
        ...current,
        ward_id: value,
        bed_id: "",
      }))

      loadBeds(value)
    }
  }

  // =======================================================
  // SUBMIT
  // =======================================================

  async function handleSubmit(event) {
    event.preventDefault()

    setError("")
    setSuccessPatient(null)

    // -----------------------------------------------------
    // Name validation
    // -----------------------------------------------------

    if (
      !form.first_name.trim() ||
      !form.last_name.trim()
    ) {
      setError(
        "First name and last name are required.",
      )
      return
    }

    // -----------------------------------------------------
    // Doctor validation
    // -----------------------------------------------------

    if (!form.doctor_id) {
      setError(
        "Please select an attending doctor.",
      )
      return
    }

    const selectedDoctor = doctors.find(
      (doctor) =>
        String(doctor.id) ===
        String(form.doctor_id),
    )

    if (!selectedDoctor) {
      setError(
        "The selected doctor is no longer available. Please select an active registered doctor.",
      )
      return
    }

    // -----------------------------------------------------
    // Admission validation
    // -----------------------------------------------------

    if (form.admit_patient) {
      if (!form.ward_id) {
        setError(
          "Please select a ward for admission.",
        )
        return
      }

      if (!form.bed_id) {
        setError(
          "Please select an available bed.",
        )
        return
      }
    }

    setLoading(true)

    try {
      const payload = {
        first_name: form.first_name.trim(),

        last_name: form.last_name.trim(),

        date_of_birth:
          form.date_of_birth || null,

        gender:
          form.gender.trim() || null,

        weight_kg:
          form.weight_kg
            ? Number(form.weight_kg)
            : null,

        doctor_id:
          Number(form.doctor_id),

        is_admitted:
          Boolean(form.admit_patient),

        ward_id:
          form.admit_patient &&
          form.ward_id
            ? Number(form.ward_id)
            : null,

        bed_id:
          form.admit_patient &&
          form.bed_id
            ? Number(form.bed_id)
            : null,
      }

      const response = await api.post(
        "/api/hospital/patients",
        payload,
      )

      const createdPatient =
        response.data?.patient

      setSuccessPatient(
        createdPatient || null,
      )

      // ---------------------------------------------------
      // Reset
      // ---------------------------------------------------

      setForm({
        first_name: "",
        last_name: "",
        date_of_birth: "",
        gender: "",
        weight_kg: "",
        doctor_id: "",
        admit_patient: false,
        ward_id: "",
        bed_id: "",
      })

      setBeds([])

    } catch (requestError) {
      console.error(
        "Create hospital patient error:",
        requestError,
      )

      setError(
        requestError?.response?.data?.detail ||
          "Unable to create the patient record.",
      )

    } finally {
      setLoading(false)
    }
  }

  // =======================================================
  // CANCEL
  // =======================================================

  function handleCancel() {
    navigate("/hospital/dashboard")
  }

  // =======================================================
  // SELECTED WARD / BED
  // =======================================================

  const selectedWard = wards.find(
    (ward) =>
      String(ward.id) ===
      String(form.ward_id),
  )

  const selectedBed = beds.find(
    (bed) =>
      String(bed.id) ===
      String(form.bed_id),
  )

  // =======================================================
  // SELECTED DOCTOR
  // =======================================================

  const selectedDoctor = doctors.find(
    (doctor) =>
      String(doctor.id) ===
      String(form.doctor_id),
  )

  // =======================================================
  // RENDER
  // =======================================================

  return (
    <MainLayout title="Register Patient">

      <div className="mx-auto max-w-5xl space-y-6">

        {/* =================================================
            HEADER
        ================================================= */}

        <div>

          <button
            type="button"
            onClick={handleCancel}
            className="mb-3 inline-flex items-center gap-2 text-sm font-semibold text-slate-500 transition hover:text-slate-900"
          >
            <ArrowLeft className="h-4 w-4" />
            Back to Hospital Dashboard
          </button>

          <p className="text-sm font-semibold text-cyan-600">
            Hospital Data Portal
          </p>

          <h1 className="mt-1 text-3xl font-bold tracking-tight text-slate-900">
            Register Patient
          </h1>

          <p className="mt-2 max-w-2xl text-sm leading-6 text-slate-500">
            Create a patient record in the shared
            SmartMedChart database.
          </p>

        </div>

        {/* =================================================
            ERROR
        ================================================= */}

        {error && (
          <div className="rounded-2xl border border-red-200 bg-red-50 p-4">

            <div className="flex items-start gap-3">

              <ClipboardPlus className="mt-0.5 h-5 w-5 shrink-0 text-red-600" />

              <div>

                <p className="font-semibold text-red-800">
                  Patient registration failed
                </p>

                <p className="mt-1 text-sm leading-6 text-red-700">
                  {error}
                </p>

              </div>

            </div>

          </div>
        )}

        {/* =================================================
            SUCCESS
        ================================================= */}

        {successPatient && (
          <div className="rounded-2xl border border-emerald-200 bg-emerald-50 p-5 shadow-sm">

            <div className="flex items-start gap-3">

              <CheckCircle2 className="mt-0.5 h-6 w-6 shrink-0 text-emerald-600" />

              <div className="flex-1">

                <p className="font-semibold text-emerald-900">
                  Patient registered successfully

                  Patient ID:

                    {successPatient.patient_id}

                <div className="mt-4 flex flex-wrap gap-2">

                  <button
                    type="button"
                    onClick={() =>
                      navigate(
                        "/hospital/dashboard",
                      )
                    }
                    className="rounded-xl bg-emerald-600 px-4 py-2.5 text-sm font-semibold text-white hover:bg-emerald-700"
                  >
                    View Patients

// PROTECTED ROUTE

function ProtectedRoute({
  children,
}) {
  const {
    user,
    loading,
  } = useAuth()

  if (loading) {
    return (
      <div className="flex min-h-screen items-center justify-center bg-slate-950 text-white">
        Loading SmartMedChart...
      </div>
    )
  }

  if (!user) {
    return (
      <Navigate
        to="/login"
        replace
      />
    )
  }

  return children
}

// =====================================================
// ROLE ROUTE
// =====================================================

function RoleRoute({
  allowedRoles,
  children,
}) {
  const {
    user,
  } = useAuth()

  const role =
    typeof user?.role === "string"
      ? user.role
          .trim()
          .toLowerCase()
      : ""

  if (!allowedRoles.includes(role)) {
    return (
      <Navigate
        to="/"
        replace
      />
    )
  }

  return children
}

// =====================================================
// HOME REDIRECT

function HomeRedirect() {
  const {
    user,
  } = useAuth()

  if (!user) {
    return (
      <Navigate
        to="/login"
        replace
      />
    )
  }

  const role =
    typeof user.role === "string"
      ? user.role
          .trim()
          .toLowerCase()
      : ""

  if (role === "nurse") {
    return (
      <Navigate
        to="/nurse/dashboard"
        replace
      />
    )
  }

  if (role === "doctor") {
    return (
      <Navigate
        to="/doctor/dashboard"
        replace
      />
    )
  }

  if (
    role === "admin" ||
    role === "administrator"
  ) {
    return (
      <Navigate
        to="/admin/dashboard"
        replace
      />
    )
  }

  if (
    role === "hospital" ||
    role === "hospital_staff"
  ) {
    return (
      <Navigate
        to="/hospital/dashboard"
        replace
      />
    )
  }

  return (
    <div className="flex min-h-screen items-center justify-center bg-slate-100 p-6">
      <div className="rounded-2xl border border-slate-200 bg-white p-8 shadow-sm">
        <h1 className="text-2xl font-bold text-slate-900">
          SmartMedChart
        </h1>

        <p className="mt-3 text-slate-600">
          Logged in, but no dashboard is configured
          for this role.
        </p>

        <pre className="mt-4 rounded-lg bg-slate-100 p-4 text-sm">
          {JSON.stringify(
            user,
            null,
            2,
          )}
        </pre>
      </div>
    </div>
  )
}
