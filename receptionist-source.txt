import {
  ArrowLeft,
  BedDouble,
  CalendarDays,
  CheckCircle2,
  ClipboardPlus,
  LoaderCircle,
  Stethoscope,
  UserPlus,
  Weight,
} from "lucide-react"

import { useEffect, useState } from "react"
import { useNavigate } from "react-router-dom"

import MainLayout from "../../components/layout/MainLayout"
import api from "../../services/api"
import {
  getHospitalDoctors,
  getWardBeds,
  getWards,
} from "../../services/admin"


// =========================================================
// MAIN
// =========================================================

export default function AddPatient() {
  const navigate = useNavigate()

  // =======================================================
  // FORM
  // =======================================================

  const [form, setForm] = useState({
    first_name: "",
    last_name: "",
    date_of_birth: "",
    gender: "",
    weight_kg: "",
    doctor_id: "",
    admit_patient: false,
    ward_id: "",
    bed_id: "",
  })

  // =======================================================
  // UI STATE
  // =======================================================

  const [loading, setLoading] = useState(false)
  const [error, setError] = useState("")
  const [successPatient, setSuccessPatient] = useState(null)

  // =======================================================
  // DOCTOR STATE
  // =======================================================

  const [doctors, setDoctors] = useState([])
  const [doctorsLoading, setDoctorsLoading] = useState(false)

  // =======================================================
  // WARD / BED STATE
  // =======================================================

  const [wards, setWards] = useState([])
  const [beds, setBeds] = useState([])

  const [wardsLoading, setWardsLoading] = useState(false)
  const [bedsLoading, setBedsLoading] = useState(false)

  // =======================================================
  // LOAD DOCTORS
  // =======================================================

  async function loadDoctors() {
    setDoctorsLoading(true)
    setError("")

    try {
      const response = await getHospitalDoctors()

      const doctorList = Array.isArray(response?.doctors)
        ? response.doctors
        : []

      setDoctors(doctorList)

    } catch (requestError) {
      console.error(
        "Load doctors error:",
        requestError,
      )

      setDoctors([])

      setError(
        requestError?.response?.data?.detail ||
          "Unable to load registered doctors.",
      )

    } finally {
      setDoctorsLoading(false)
    }
  }

  // =======================================================
  // LOAD WARDS
  // =======================================================

  async function loadWards() {
    setWardsLoading(true)

    try {
      const response = await getWards()

      setWards(
        Array.isArray(response?.wards)
          ? response.wards
          : [],
      )

    } catch (requestError) {
      console.error(
        "Load wards error:",
        requestError,
      )

      setError(
        requestError?.response?.data?.detail ||
          "Unable to load hospital wards.",
      )

    } finally {
      setWardsLoading(false)
    }
  }

  // =======================================================
  // LOAD BEDS
  // =======================================================

  async function loadBeds(wardId) {
    if (!wardId) {
      setBeds([])
      return
    }

    setBedsLoading(true)

    try {
      const response = await getWardBeds(wardId)

      const availableBeds =
        Array.isArray(response?.beds)
          ? response.beds.filter(
              (bed) =>
                bed.is_active &&
                !bed.is_occupied,
            )
          : []

      setBeds(availableBeds)

    } catch (requestError) {
      console.error(
        "Load beds error:",
        requestError,
      )

      setBeds([])

      setError(
        requestError?.response?.data?.detail ||
          "Unable to load beds for this ward.",
      )

    } finally {
      setBedsLoading(false)
    }
  }

  // =======================================================
  // INITIAL LOAD
  // =======================================================

  useEffect(() => {
    loadDoctors()
    loadWards()
  }, [])

  // =======================================================
  // HANDLE INPUT
  // =======================================================

  function handleChange(event) {
    const {
      name,
      value,
      type,
      checked,
    } = event.target

    setError("")

    setForm((current) => ({
      ...current,
      [name]:
        type === "checkbox"
          ? checked
          : value,
    }))

    // -----------------------------------------------------
    // Disable admission
    // -----------------------------------------------------

    if (
      name === "admit_patient" &&
      !checked
    ) {
      setForm((current) => ({
        ...current,
        admit_patient: false,
        ward_id: "",
        bed_id: "",
      }))

      setBeds([])
    }

    // -----------------------------------------------------
    // Ward changed
    // -----------------------------------------------------

    if (name === "ward_id") {
      setForm((current) => ({
        ...current,
        ward_id: value,
        bed_id: "",
      }))

      loadBeds(value)
    }
  }

  // =======================================================
  // SUBMIT
  // =======================================================

  async function handleSubmit(event) {
    event.preventDefault()

    setError("")
    setSuccessPatient(null)

    // -----------------------------------------------------
    // Name validation
    // -----------------------------------------------------

    if (
      !form.first_name.trim() ||
      !form.last_name.trim()
    ) {
      setError(
        "First name and last name are required.",
      )
      return
    }

    // -----------------------------------------------------
    // Doctor validation
    // -----------------------------------------------------

    if (!form.doctor_id) {
      setError(
        "Please select an attending doctor.",
      )
      return
    }

    const selectedDoctor = doctors.find(
      (doctor) =>
        String(doctor.id) ===
        String(form.doctor_id),
    )

    if (!selectedDoctor) {
      setError(
        "The selected doctor is no longer available. Please select an active registered doctor.",
      )
      return
    }

    // -----------------------------------------------------
    // Admission validation
    // -----------------------------------------------------

    if (form.admit_patient) {
      if (!form.ward_id) {
        setError(
          "Please select a ward for admission.",
        )
        return
      }

      if (!form.bed_id) {
        setError(
          "Please select an available bed.",
        )
        return
      }
    }

    setLoading(true)

    try {
      const payload = {
        first_name: form.first_name.trim(),

        last_name: form.last_name.trim(),

        date_of_birth:
          form.date_of_birth || null,

        gender:
          form.gender.trim() || null,

        weight_kg:
          form.weight_kg
            ? Number(form.weight_kg)
            : null,

        doctor_id:
          Number(form.doctor_id),

        is_admitted:
          Boolean(form.admit_patient),

        ward_id:
          form.admit_patient &&
          form.ward_id
            ? Number(form.ward_id)
            : null,

        bed_id:
          form.admit_patient &&
          form.bed_id
            ? Number(form.bed_id)
            : null,
      }

      const response = await api.post(
        "/api/hospital/patients",
        payload,
      )

      const createdPatient =
        response.data?.patient

      setSuccessPatient(
        createdPatient || null,
      )

      // ---------------------------------------------------
      // Reset
      // ---------------------------------------------------

      setForm({
        first_name: "",
        last_name: "",
        date_of_birth: "",
        gender: "",
        weight_kg: "",
        doctor_id: "",
        admit_patient: false,
        ward_id: "",
        bed_id: "",
      })

      setBeds([])

    } catch (requestError) {
      console.error(
        "Create hospital patient error:",
        requestError,
      )

      setError(
        requestError?.response?.data?.detail ||
          "Unable to create the patient record.",
      )

    } finally {
      setLoading(false)
    }
  }

  // =======================================================
  // CANCEL
  // =======================================================

  function handleCancel() {
    navigate("/hospital/dashboard")
  }

  // =======================================================
  // SELECTED WARD / BED
  // =======================================================

  const selectedWard = wards.find(
    (ward) =>
      String(ward.id) ===
      String(form.ward_id),
  )

  const selectedBed = beds.find(
    (bed) =>
      String(bed.id) ===
      String(form.bed_id),
  )

  // =======================================================
  // SELECTED DOCTOR
  // =======================================================

  const selectedDoctor = doctors.find(
    (doctor) =>
      String(doctor.id) ===
      String(form.doctor_id),
  )

  // =======================================================
  // RENDER
  // =======================================================

  return (
    <MainLayout title="Register Patient">

      <div className="mx-auto max-w-5xl space-y-6">

        {/* =================================================
            HEADER
        ================================================= */}

        <div>

          <button
            type="button"
            onClick={handleCancel}
            className="mb-3 inline-flex items-center gap-2 text-sm font-semibold text-slate-500 transition hover:text-slate-900"
          >
            <ArrowLeft className="h-4 w-4" />
            Back to Hospital Dashboard
          </button>

          <p className="text-sm font-semibold text-cyan-600">
            Hospital Data Portal
          </p>

          <h1 className="mt-1 text-3xl font-bold tracking-tight text-slate-900">
            Register Patient
          </h1>

          <p className="mt-2 max-w-2xl text-sm leading-6 text-slate-500">
            Create a patient record in the shared
            SmartMedChart database.
          </p>

        </div>

        {/* =================================================
            ERROR
        ================================================= */}

        {error && (
          <div className="rounded-2xl border border-red-200 bg-red-50 p-4">

            <div className="flex items-start gap-3">

              <ClipboardPlus className="mt-0.5 h-5 w-5 shrink-0 text-red-600" />

              <div>

                <p className="font-semibold text-red-800">
                  Patient registration failed
                </p>

                <p className="mt-1 text-sm leading-6 text-red-700">
                  {error}
                </p>

              </div>

            </div>

          </div>
        )}

        {/* =================================================
            SUCCESS
        ================================================= */}

        {successPatient && (
          <div className="rounded-2xl border border-emerald-200 bg-emerald-50 p-5 shadow-sm">

            <div className="flex items-start gap-3">

              <CheckCircle2 className="mt-0.5 h-6 w-6 shrink-0 text-emerald-600" />

              <div className="flex-1">

                <p className="font-semibold text-emerald-900">
                  Patient registered successfully
                </p>

                <p className="mt-1 text-sm text-emerald-800">
                  Patient ID:
                  {" "}
                  <span className="font-bold">
                    {successPatient.patient_id}
                  </span>
                </p>

                <div className="mt-4 flex flex-wrap gap-2">

                  <button
                    type="button"
                    onClick={() =>
                      navigate(
                        "/hospital/dashboard",
                      )
                    }
                    className="rounded-xl bg-emerald-600 px-4 py-2.5 text-sm font-semibold text-white hover:bg-emerald-700"
                  >
                    View Patients
                  </button>

                  <button
                    type="button"
                    onClick={() =>
                      setSuccessPatient(null)
                    }
                    className="rounded-xl border border-slate-200 bg-white px-4 py-2.5 text-sm font-semibold text-slate-700 hover:bg-slate-50"
                  >
                    Register Another
                  </button>

                </div>

              </div>

            </div>

          </div>
        )}

        {/* =================================================
            FORM
        ================================================= */}

        <form
          onSubmit={handleSubmit}
          className="overflow-hidden rounded-3xl border border-slate-200 bg-white shadow-sm"
        >

          {/* HEADER */}

          <div className="border-b border-slate-200 bg-slate-50 p-5 md:p-6">

            <div className="flex items-center gap-3">

              <div className="rounded-xl bg-cyan-100 p-3">
                <UserPlus className="h-5 w-5 text-cyan-700" />
              </div>

              <div>

                <h2 className="font-bold text-slate-900">
                  Patient Information
                </h2>

                <p className="mt-1 text-sm text-slate-500">
                  Enter patient and attending doctor details.
                </p>

              </div>

            </div>

          </div>

          <div className="space-y-8 p-5 md:p-6">

            {/* =================================================
                BASIC INFORMATION
            ================================================= */}

            <section>

              <div className="mb-4">

                <h3 className="font-semibold text-slate-900">
                  Basic Information
                </h3>

                <p className="mt-1 text-sm text-slate-500">
                  Identity and demographic information.
                </p>

              </div>

              <div className="grid gap-5 md:grid-cols-2">

                {/* Patient ID */}

                <div>

                  <label className="mb-2 block text-sm font-semibold text-slate-700">
                    Patient ID
                  </label>

                  <input
                    type="text"
                    value="Auto-generated"
                    disabled
                    className="w-full rounded-xl border border-emerald-200 bg-emerald-50 px-4 py-3 text-sm font-semibold text-emerald-700"
                  />

                  <p className="mt-2 text-xs text-slate-500">
                    Assigned automatically after registration.
                  </p>

                </div>

                {/* First Name */}

                <div>

                  <label
                    htmlFor="first_name"
                    className="mb-2 block text-sm font-semibold text-slate-700"
                  >
                    First Name
                    <span className="ml-1 text-red-500">
                      *
                    </span>
                  </label>

                  <input
                    id="first_name"
                    name="first_name"
                    value={form.first_name}
                    onChange={handleChange}
                    placeholder="Enter first name"
                    required
                    maxLength={100}
                    className="w-full rounded-xl border border-slate-200 px-4 py-3 text-sm outline-none focus:border-cyan-400 focus:ring-2 focus:ring-cyan-100"
                  />

                </div>

                {/* Last Name */}

                <div>

                  <label
                    htmlFor="last_name"
                    className="mb-2 block text-sm font-semibold text-slate-700"
                  >
                    Last Name
                    <span className="ml-1 text-red-500">
                      *
                    </span>
                  </label>

                  <input
                    id="last_name"
                    name="last_name"
                    value={form.last_name}
                    onChange={handleChange}
                    placeholder="Enter last name"
                    required
                    maxLength={100}
                    className="w-full rounded-xl border border-slate-200 px-4 py-3 text-sm outline-none focus:border-cyan-400 focus:ring-2 focus:ring-cyan-100"
                  />

                </div>

                {/* Date */}

                <div>

                  <label
                    htmlFor="date_of_birth"
                    className="mb-2 flex items-center gap-2 text-sm font-semibold text-slate-700"
                  >
                    <CalendarDays className="h-4 w-4 text-slate-400" />
                    Date of Birth
                  </label>

                  <input
                    id="date_of_birth"
                    name="date_of_birth"
                    type="date"
                    value={form.date_of_birth}
                    onChange={handleChange}
                    className="w-full rounded-xl border border-slate-200 px-4 py-3 text-sm outline-none focus:border-cyan-400 focus:ring-2 focus:ring-cyan-100"
                  />

                </div>

                {/* Gender */}

                <div>

                  <label
                    htmlFor="gender"
                    className="mb-2 block text-sm font-semibold text-slate-700"
                  >
                    Gender
                  </label>

                  <select
                    id="gender"
                    name="gender"
                    value={form.gender}
                    onChange={handleChange}
                    className="w-full rounded-xl border border-slate-200 px-4 py-3 text-sm outline-none focus:border-cyan-400 focus:ring-2 focus:ring-cyan-100"
                  >

                    <option value="">
                      Select gender
                    </option>

                    <option value="Male">
                      Male
                    </option>

                    <option value="Female">
                      Female
                    </option>

                    <option value="Other">
                      Other
                    </option>

                  </select>

                </div>

                {/* Weight */}

                <div>

                  <label
                    htmlFor="weight_kg"
                    className="mb-2 flex items-center gap-2 text-sm font-semibold text-slate-700"
                  >
                    <Weight className="h-4 w-4 text-slate-400" />
                    Weight (kg)
                  </label>

                  <input
                    id="weight_kg"
                    name="weight_kg"
                    type="number"
                    min="0.1"
                    max="500"
                    step="0.1"
                    value={form.weight_kg}
                    onChange={handleChange}
                    placeholder="e.g. 72.5"
                    className="w-full rounded-xl border border-slate-200 px-4 py-3 text-sm outline-none focus:border-cyan-400 focus:ring-2 focus:ring-cyan-100"
                  />

                </div>

                {/* =================================================
                    DOCTOR
                ================================================= */}

                <div className="md:col-span-2">

                  <label
                    htmlFor="doctor_id"
                    className="mb-2 flex items-center gap-2 text-sm font-semibold text-slate-700"
                  >

                    <Stethoscope className="h-4 w-4 text-cyan-600" />

                    Attending Doctor

                    <span className="text-red-500">
                      *
                    </span>

                  </label>

                  <select
                    id="doctor_id"
                    name="doctor_id"
                    value={form.doctor_id}
                    onChange={handleChange}
                    required
                    disabled={doctorsLoading}
                    className="w-full rounded-xl border border-cyan-200 bg-white px-4 py-3 text-sm outline-none focus:border-cyan-400 focus:ring-2 focus:ring-cyan-100 disabled:cursor-not-allowed disabled:bg-slate-100"
                  >

                    <option value="">
                      {doctorsLoading
                        ? "Loading registered doctors..."
                        : doctors.length === 0
                          ? "No active doctors registered"
                          : "Select attending doctor"}
                    </option>

                    {doctors.map(
                      (doctor) => (
                        <option
                          key={doctor.id}
                          value={doctor.id}
                        >
                          {doctor.full_name}
                          {" • "}
                          {doctor.username}
                        </option>
                      ),
                    )}

                  </select>

                  <p className="mt-2 text-xs text-slate-500">
                    Doctors are loaded automatically from
                    active Doctor accounts in Staff Management.
                  </p>

                  {selectedDoctor && (
                    <div className="mt-3 rounded-xl border border-cyan-200 bg-cyan-50 p-3">

                      <p className="text-xs font-bold uppercase tracking-wide text-cyan-700">
                        Selected Doctor
                      </p>

                      <p className="mt-1 font-semibold text-cyan-900">
                        {selectedDoctor.full_name}
                      </p>

                      <p className="mt-1 text-xs text-cyan-700">
                        Username:
                        {" "}
                        {selectedDoctor.username}
                      </p>

                    </div>
                  )}

                </div>

              </div>

            </section>

            {/* =================================================
                ADMISSION
            ================================================= */}

            <section className="border-t border-slate-100 pt-8">

              <div className="rounded-2xl border border-slate-200 bg-slate-50 p-4">

                <label className="flex cursor-pointer items-start gap-3">

                  <input
                    type="checkbox"
                    name="admit_patient"
                    checked={form.admit_patient}
                    onChange={handleChange}
                    className="mt-1 h-4 w-4 rounded border-slate-300 text-cyan-600 focus:ring-cyan-500"
                  />

                  <span>

                    <span className="block text-sm font-semibold text-slate-800">
                      Admit Patient
                    </span>

                    <span className="mt-1 block text-sm leading-5 text-slate-500">
                      Select this option to assign the patient
                      to a real ward and available bed.
                    </span>

                  </span>

                </label>

              </div>

              {/* Ward / Bed */}

              {form.admit_patient && (

                <div className="mt-5 rounded-2xl border border-cyan-100 bg-cyan-50/40 p-5">

                  <div className="mb-5 flex items-center gap-3">

                    <div className="rounded-xl bg-cyan-100 p-2">

                      <BedDouble className="h-5 w-5 text-cyan-700" />

                    </div>

                    <div>

                      <h3 className="font-semibold text-slate-900">
                        Hospital Location
                      </h3>

                      <p className="mt-1 text-sm text-slate-500">
                        Select the ward first, then choose
                        an available bed.
                      </p>

                    </div>

                  </div>

                  <div className="grid gap-5 md:grid-cols-2">

                    {/* Ward */}

                    <div>

                      <label
                        htmlFor="ward_id"
                        className="mb-2 block text-sm font-semibold text-slate-700"
                      >
                        Ward
                        <span className="ml-1 text-red-500">
                          *
                        </span>
                      </label>

                      <select
                        id="ward_id"
                        name="ward_id"
                        value={form.ward_id}
                        onChange={handleChange}
                        required
                        disabled={wardsLoading}
                        className="w-full rounded-xl border border-slate-200 bg-white px-4 py-3 text-sm outline-none focus:border-cyan-400 focus:ring-2 focus:ring-cyan-100 disabled:bg-slate-100"
                      >

                        <option value="">
                          {wardsLoading
                            ? "Loading wards..."
                            : "Select a ward"}
                        </option>

                        {wards.map(
                          (ward) => (
                            <option
                              key={ward.id}
                              value={ward.id}
                              disabled={
                                Number(
                                  ward.available_beds,
                                ) === 0
                              }
                            >
                              {ward.name}
                              {" • "}
                              {ward.available_beds}
                              {" available"}
                            </option>
                          ),
                        )}

                      </select>

                    </div>

                    {/* Bed */}

                    <div>

                      <label
                        htmlFor="bed_id"
                        className="mb-2 block text-sm font-semibold text-slate-700"
                      >
                        Available Bed
                        <span className="ml-1 text-red-500">
                          *
                        </span>
                      </label>

                      <select
                        id="bed_id"
                        name="bed_id"
                        value={form.bed_id}
                        onChange={handleChange}
                        required
                        disabled={
                          !form.ward_id ||
                          bedsLoading
                        }
                        className="w-full rounded-xl border border-slate-200 bg-white px-4 py-3 text-sm outline-none focus:border-cyan-400 focus:ring-2 focus:ring-cyan-100 disabled:bg-slate-100"
                      >

                        <option value="">
                          {!form.ward_id
                            ? "Select ward first"
                            : bedsLoading
                              ? "Loading beds..."
                              : beds.length === 0
                                ? "No available beds"
                                : "Select available bed"}
                        </option>

                        {beds.map(
                          (bed) => (
                            <option
                              key={bed.id}
                              value={bed.id}
                            >
                              {bed.bed_number}
                              {" • "}
                              Available
                            </option>
                          ),
                        )}

                      </select>

                    </div>

                  </div>

                  {/* Selected location */}

                  {selectedWard && selectedBed && (

                    <div className="mt-5 rounded-xl border border-emerald-200 bg-emerald-50 p-4">

                      <p className="text-xs font-bold uppercase tracking-wide text-emerald-700">
                        Selected Location
                      </p>

                      <p className="mt-1 font-semibold text-emerald-900">
                        {selectedWard.name}
                        {" • "}
                        {selectedBed.bed_number}
                      </p>

                    </div>

                  )}

                </div>
              )}

            </section>

          </div>

          {/* =================================================
              ACTIONS
          ================================================= */}

          <div className="flex flex-col-reverse gap-3 border-t border-slate-200 bg-slate-50 p-5 sm:flex-row sm:justify-end md:p-6">

            <button
              type="button"
              onClick={handleCancel}
              disabled={loading}
              className="rounded-xl border border-slate-200 bg-white px-5 py-3 text-sm font-semibold text-slate-700 hover:bg-slate-100 disabled:opacity-50"
            >
              Cancel
            </button>

            <button
              type="submit"
              disabled={
                loading ||
                doctorsLoading ||
                doctors.length === 0
              }
              className="inline-flex items-center justify-center gap-2 rounded-xl bg-cyan-600 px-5 py-3 text-sm font-semibold text-white hover:bg-cyan-700 disabled:cursor-not-allowed disabled:opacity-60"
            >

              {loading ? (
                <>
                  <LoaderCircle className="h-4 w-4 animate-spin" />
                  Registering...
                </>
              ) : (
                <>
                  <UserPlus className="h-4 w-4" />
                  Register Patient
                </>
              )}

            </button>

          </div>

        </form>

        {/* =================================================
            FOOTER
        ================================================= */}

        <div className="rounded-2xl border border-slate-200 bg-white p-5 shadow-sm">

          <p className="text-sm font-semibold text-slate-800">
            Patient registration
          </p>

          <p className="mt-1 text-sm leading-6 text-slate-500">
            Patient records are stored in the SmartMedChart
            shared database and are available to authorized
            clinical users.
          </p>

        </div>

      </div>

    </MainLayout>
  )
}

function todayString() {
  const d = new Date()
  const month = String(d.getMonth() + 1).padStart(2, "0")
  const day = String(d.getDate()).padStart(2, "0")
  return `${d.getFullYear()}-${month}-${day}`
}

  appointment_date: todayString(),

export default function CareMatch() {
  const [form, setForm] = useState(emptyForm)
  const [recommendations, setRecommendations] = useState([])
  const [selectedDoctorId, setSelectedDoctorId] = useState("")
  const [searching, setSearching] = useState(false)
  const [submitting, setSubmitting] = useState(false)
  const [error, setError] = useState("")
  const [notice, setNotice] = useState("")
  const [success, setSuccess] = useState(null)
  const [pendingRegistration, setPendingRegistration] = useState(null)

  const selectedDoctor = recommendations.find(
    (doctor) => String(doctor.doctor_id) === String(selectedDoctorId),
  )

  function updateField(event) {
    const { name, value } = event.target

    setForm((current) => ({ ...current, [name]: value }))
    setError("")
    setNotice("")
    setSuccess(null)

    if (name === "problem_category" || name === "visit_notes") {
      setRecommendations([])
      setSelectedDoctorId("")
    }
  }

  function getVisitReason() {
    return [
      form.problem_category,
      form.visit_notes.trim(),
    ].filter(Boolean).join(". ")
  }

  async function recommendDoctors() {
    if (!form.problem_category) {
      setError("Select the patient's main problem first.")
      return
    }

    setSearching(true)
    setError("")
    setNotice("")
    setRecommendations([])
    setSelectedDoctorId("")

    try {
      const response = await api.post("/api/carematch/recommendations", {
        visit_reason: getVisitReason(),
      })

      const list = Array.isArray(response.data?.recommendations)
        ? response.data.recommendations
        : []

      setRecommendations(list)

      if (list.length) {
        setSelectedDoctorId(String(list[0].doctor_id))
        setNotice(
          "Review the available doctors and choose the appropriate doctor. Suggestions are based on profile keywords, not a diagnosis.",
        )
      } else {
        setNotice(
          response.data?.message ||
          "No matching available doctor profiles were found. Ask an administrator to update doctor profiles.",
        )
      }
    } catch (requestError) {
      setError(
        requestError?.response?.data?.detail ||
        "Could not find doctor suggestions. Check the backend connection and CareMatch API.",
      )
    } finally {
      setSearching(false)
    }
  }

  async function saveAppointment(registration) {
    try {
      const response = await api.post("/api/carematch/appointments", {
        patient_id: registration.patient.id,
        encounter_id: registration.encounter.id,
        doctor_id: registration.doctor_id,
        visit_reason: registration.visit_reason,
        appointment_date: registration.appointment_date,
        appointment_time: registration.appointment_time,
      })

      setSuccess({
        patient: registration.patient,
        doctorName: registration.doctor_name,
        appointment: response.data?.appointment,
      })

      setPendingRegistration(null)
      setForm(emptyForm())
      setRecommendations([])
      setSelectedDoctorId("")
      setNotice("")
      return true
    } catch (requestError) {
      setPendingRegistration(registration)
      setError(
        `Patient ${registration.patient.patient_id} has already been registered, but the appointment could not be saved. ${requestError?.response?.data?.detail || "Check the backend and MongoDB, then retry."}`,
      )
      return false
    }
  }

  async function handleSubmit(event) {
    event.preventDefault()
    setError("")
    setSuccess(null)

    if (pendingRegistration) {
      setSubmitting(true)
      await saveAppointment(pendingRegistration)
      setSubmitting(false)
      return
    }

    if (!form.first_name.trim() || !form.last_name.trim()) {
      setError("Enter the patient's first and last name.")
      return
    }

    if (!form.problem_category) {
      setError("Select the patient's main problem.")
      return
    }

    if (!selectedDoctor) {
      setError("Select a suggested doctor before registering the patient.")
      return
    }

    if (!form.appointment_date || !form.appointment_time) {
      setError("Choose an appointment date and time.")
      return
    }

    if (form.appointment_date < todayString()) {
      setError("The appointment date cannot be in the past.")
      return
    }

    setSubmitting(true)

    try {
      const patientResponse = await api.post("/api/hospital/patients", {
        first_name: form.first_name.trim(),
        last_name: form.last_name.trim(),
        date_of_birth: form.date_of_birth || null,
        gender: form.gender || null,
        weight_kg: form.weight_kg ? Number(form.weight_kg) : null,
        doctor_id: Number(selectedDoctor.doctor_id),
        is_admitted: false,
        ward_id: null,
        bed_id: null,
      })

      const patient = patientResponse.data?.patient
      const encounter = patientResponse.data?.encounter

      if (!patient?.id || !encounter?.id) {
        setError(
          "The patient API did not return both the patient and encounter IDs. Check its response before trying again to avoid duplicate records.",
        )
        return
      }

      const registration = {
        patient,
        encounter,
        doctor_id: Number(selectedDoctor.doctor_id),
        doctor_name: selectedDoctor.full_name,
        visit_reason: getVisitReason(),
        appointment_date: form.appointment_date,
        appointment_time: form.appointment_time,
      }

      setPendingRegistration(registration)
      await saveAppointment(registration)
    } catch (requestError) {
      setError(
        requestError?.response?.data?.detail ||
        "Patient registration failed. Check the server message before trying again.",
      )
    } finally {
      setSubmitting(false)
    }
  }

  return (
    <MainLayout title="Receptionist Counter">
      <div className="mx-auto max-w-7xl space-y-6 pb-8">

        <header className="relative overflow-hidden rounded-3xl bg-gradient-to-br from-slate-950 via-indigo-950 to-indigo-800 p-6 text-white shadow-lg sm:p-8">
          <div className="absolute -right-12 -top-20 h-64 w-64 rounded-full border-[35px] border-white/5" />
          <div className="relative flex flex-col justify-between gap-6 md:flex-row md:items-center">
            <div>
              <div className="inline-flex items-center gap-2 rounded-full border border-white/15 bg-white/10 px-3 py-1.5 text-xs font-semibold text-indigo-100">
                <Activity size={14} />
                SMARTMEDCHART · CAREMATCH
              </div>
              <h1 className="mt-4 text-3xl font-bold tracking-tight sm:text-4xl">
                Receptionist Counter
              </h1>
              <p className="mt-3 max-w-2xl text-sm leading-6 text-indigo-100">
                Register a patient, select their main concern, and review available doctors
                matched against their configured specializations and expertise.
              </p>
            </div>
            <div className="flex h-20 w-20 shrink-0 items-center justify-center rounded-2xl border border-white/15 bg-white/10">
              <Stethoscope size={38} />
            </div>
          </div>
        </header>

        {error && (
          <div role="alert" className="flex gap-3 rounded-2xl border border-rose-200 bg-rose-50 p-4 text-sm text-rose-800">
            <AlertCircle size={19} className="mt-0.5 shrink-0" />
            <span>{error}</span>
          </div>
        )}

        {notice && !error && (
          <div role="status" className="rounded-2xl border border-indigo-100 bg-indigo-50 p-4 text-sm text-indigo-900">
            {notice}
          </div>
        )}

        {success && (
          <div role="status" className="rounded-2xl border border-emerald-200 bg-emerald-50 p-5">
            <div className="flex items-start gap-3">
              <CheckCircle2 className="mt-0.5 shrink-0 text-emerald-700" size={24} />
              <div>
                <h2 className="font-bold text-emerald-950">Registration complete</h2>
                <p className="mt-1 text-sm text-emerald-900">
                  Patient ID: <strong>{success.patient.patient_id}</strong>
                </p>
                <p className="mt-1 text-sm text-emerald-900">

                  </p>

            Patient <strong>{pendingRegistration.patient.patient_id}</strong> already exists.

def require_admin_permission(current_user: User):
    permission_names = {
        permission.name
        for permission in current_user.role.permissions
    }

    if "MANAGE_USERS" not in permission_names:
        raise HTTPException(
            status_code=403,
            detail="Administrator permission required.",
        )


def clean_list(values: list[str], limit: int = 30) -> list[str]:
    """Trim entries and remove duplicates."""
    cleaned = []
    seen = set()

    for value in values:
        item = value.strip()

        if not item:
            continue

        if len(item) > 100:
            raise HTTPException(
                status_code=400,
                detail="Each list entry must be 100 characters or fewer.",
            )

        key = item.casefold()

        if key not in seen:
            cleaned.append(item)
            seen.add(key)

        if len(cleaned) >= limit:
            break

    return cleaned


def profile_to_dict(profile: dict | None, doctor_id: int) -> dict:
    """Return a consistent profile structure."""
    profile = profile or {}

    return {
        "doctor_id": doctor_id,
        "qualifications": profile.get("qualifications", ""),
        "specializations": profile.get("specializations", []),
        "experience_years": profile.get("experience_years", 0),
        "expertise_keywords": profile.get("expertise_keywords", []),
        "languages": profile.get("languages", []),
        "consultation_fee": profile.get("consultation_fee"),
        "available_days": profile.get("available_days", []),
        "is_available": profile.get("is_available", True),
        "updated_at": profile.get("updated_at"),
    }


def find_active_doctor(db: Session, doctor_id: int) -> User:
    doctor = db.scalar(
        select(User)
        .join(Role, User.role_id == Role.id)
        .where(
            User.id == doctor_id,
            Role.name == "Doctor",
            User.is_active.is_(True),
        )
    )

    if doctor is None:
        raise HTTPException(
            status_code=404,
            detail="Active doctor account not found.",
        )

    return doctor

@router.get("/admin/doctors")
def get_carematch_doctors(
    db: Session = Depends(get_db),
    current_user: User = Depends(
        require_permission("MANAGE_USERS")
    ),
):
    require_admin_permission(current_user)

    doctors = db.scalars(
        select(User)
        .join(Role, User.role_id == Role.id)
        .where(
            Role.name == "Doctor",
            User.is_active.is_(True),
        )
        .order_by(User.full_name.asc())
    ).all()

    try:
        profiles = {
            int(profile["doctor_id"]): profile
            for profile in mongo_db.doctor_profiles.find({})
            if profile.get("doctor_id") is not None
        }

    except PyMongoError as exc:
        raise HTTPException(
            status_code=503,
            detail="MongoDB is unavailable. Check your MongoDB service.",
        ) from exc

    return {
        "count": len(doctors),
        "doctors": [
            {
                "id": doctor.id,
                "full_name": doctor.full_name,
                "username": doctor.username,
                "email": doctor.email,
                "profile": profile_to_dict(
                    profiles.get(doctor.id),
                    doctor.id,
                ),
                "profile_configured": doctor.id in profiles,
            }
            for doctor in doctors
        ],
    }

@router.put("/admin/doctors/{doctor_id}/profile")
def save_doctor_profile(
    doctor_id: int,
    request: DoctorProfileRequest,
    db: Session = Depends(get_db),
    current_user: User = Depends(
        require_permission("MANAGE_USERS")
    ),
):
    require_admin_permission(current_user)

    doctor = find_active_doctor(db, doctor_id)

    specializations = clean_list(request.specializations)
    expertise_keywords = clean_list(request.expertise_keywords)
    languages = clean_list(request.languages)
    available_days = clean_list(request.available_days, limit=7)

    allowed_days = {
        "Monday",
        "Tuesday",
        "Wednesday",

@router.post("/recommendations")
def recommend_doctors(
    request: dict,
    db=Depends(get_db),
    current_user: User = Depends(
        require_permission("REGISTER_PATIENT")
    ),
):
    """
    Suggest available doctors using their configured
    specializations and expertise keywords.

    This is a routing aid, not a diagnosis.
    """

    reason = str(
        request.get("visit_reason") or ""
    ).strip()

    if not reason:
        raise HTTPException(
            status_code=400,
            detail="Please select the patient's problem.",
        )

    if len(reason) > 1500:
        raise HTTPException(
            status_code=400,
            detail="Visit reason is too long.",
        )

    # Find active doctor accounts in the existing SQL database.
    doctors = db.scalars(
        select(User)
        .join(Role, User.role_id == Role.id)
        .where(
            Role.name == "Doctor",
            User.is_active.is_(True),
        )
        .order_by(User.full_name.asc())
    ).all()

    if not doctors:
        return {
            "count": 0,
            "recommendations": [],
            "message": "No active doctors are registered.",
        }

    # Use the same doctor_profiles collection as the
    # existing administrator Doctor Profiles page.
    try:
        profiles = list(
            mongo_db.doctor_profiles.find({
                "doctor_id": {
                    "$in": [doctor.id for doctor in doctors]
                },
                "is_available": True,
            })
        )
    except PyMongoError as exc:
        raise HTTPException(
            status_code=503,
            detail=(
                "MongoDB is unavailable. Check that the "
                "MongoDB service is running."
            ),
        ) from exc

    doctors_by_id = {
        doctor.id: doctor
        for doctor in doctors
    }

    reason_lower = reason.casefold()

    reason_words = set(
        re.findall(r"[a-z0-9]+", reason_lower)
    )

    recommendations = []

    for profile in profiles:
        doctor_id = profile.get("doctor_id")

        if doctor_id is None:
            continue

        doctor = doctors_by_id.get(int(doctor_id))

        if doctor is None:
            continue

        specializations = profile.get(
            "specializations", []
        )

        expertise = profile.get(
            "expertise_keywords", []
        )

        if isinstance(specializations, str):
            specializations = [
                item.strip()
                for item in specializations.split(",")
                if item.strip()
            ]

        if isinstance(expertise, str):
            expertise = [
                item.strip()
                for item in expertise.split(",")
                if item.strip()
            ]

        profile_terms = (
            specializations + expertise
        )

        matched_terms = []

        for term in profile_terms:
            term = str(term).strip()

            if not term:
                continue

            term_words = set(
                re.findall(
                    r"[a-z0-9]+",
                    term.casefold(),
                )
            )

            # A direct specialization or keyword phrase match.
            if (
                term.casefold() in reason_lower
                or reason_words.intersection(term_words)
            ):
                matched_terms.append(term)

        # Score reflects keyword overlap, not medical suitability.
        unique_matches = list(
            dict.fromkeys(matched_terms)
        )

        match_score = min(
            100,
            len(unique_matches) * 25,
        )

        if not unique_matches:
            match_reasons = [
                "No direct keyword match; review the doctor's profile."
            ]
        else:
            match_reasons = unique_matches[:5]

        recommendations.append({
            "doctor_id": doctor.id,
            "full_name": doctor.full_name,
            "qualifications": profile.get(
                "qualifications", ""
            ),
            "specializations": specializations,
            "expertise_keywords": expertise,
            "languages": profile.get(
                "languages", []
            ),
            "experience_years": profile.get(
                "experience_years", 0
            ),
            "consultation_fee": profile.get(
                "consultation_fee"
            ),
            "match_score": match_score,
            "match_reasons": match_reasons,
        })

    recommendations.sort(
        key=lambda item: (
            -item["match_score"],
            item["full_name"].casefold(),
        )
    )

    return {
        "count": len(recommendations),
        "recommendations": recommendations,
        "message": (
            None if recommendations else
            "No available doctor profiles found. "
            "Ask the administrator to configure doctor "
            "specializations and availability."
        ),
    }

@router.post("/appointments")
def save_carematch_appointment(
    request: dict,
    db=Depends(get_db),
    current_user=Depends(
        require_permission("REGISTER_PATIENT")
    ),
):
    """Save appointment metadata linked to existing SQL records."""

    try:
        patient_id = int(request.get("patient_id"))
        encounter_id = int(request.get("encounter_id"))
        doctor_id = int(request.get("doctor_id"))
    except (TypeError, ValueError):
        raise HTTPException(
            status_code=400,
            detail="Valid patient, encounter, and doctor IDs are required.",
        )

    visit_reason = str(
        request.get("visit_reason") or ""
    ).strip()

    appointment_date = str(
        request.get("appointment_date") or ""
    ).strip()

    appointment_time = str(
        request.get("appointment_time") or ""
    ).strip()

    if not visit_reason:
        raise HTTPException(
            status_code=400,
            detail="The reason for visit is required.",
        )

    if len(visit_reason) > 1500:
        raise HTTPException(
            status_code=400,
            detail="The reason for visit is too long.",
        )

    try:
        parsed_date = date.fromisoformat(appointment_date)
        parsed_time = datetime.strptime(
            appointment_time, "%H:%M"
        ).strftime("%H:%M")
    except ValueError as exc:
        raise HTTPException(
            status_code=400,
            detail="Use a valid appointment date and time.",
        ) from exc

    if parsed_date < date.today():
        raise HTTPException(
            status_code=400,
            detail="The appointment date cannot be in the past.",
        )

    # Confirm the patient exists in the existing SQL database.
    patient = db.get(Patient, patient_id)

    if patient is None:
        raise HTTPException(
            status_code=404,
            detail="Patient not found.",
        )

    # Confirm the encounter belongs to this patient.
    encounter = db.get(Encounter, encounter_id)

    if (
        encounter is None
        or encounter.patient_id != patient.id
    ):
        raise HTTPException(
            status_code=400,
            detail="The encounter does not belong to this patient.",
        )

    if (
        encounter.doctor_id != doctor_id
        or encounter.status != "ACTIVE"
    ):
        raise HTTPException(
            status_code=400,
            detail=(
                "The selected doctor does not match the active "
                "patient encounter."
            ),
        )

    # Confirm the doctor account is still active.
    doctor = db.scalar(
        select(User)
        .join(Role, User.role_id == Role.id)
        .where(
            User.id == doctor_id,
            Role.name == "Doctor",
            User.is_active.is_(True),
        )
    )

    if doctor is None:
        raise HTTPException(
            status_code=404,
            detail="The selected doctor is no longer active.",
        )

    # Recheck the doctor's current availability in MongoDB.
    try:
        profile = mongo_db.doctor_profiles.find_one({
            "doctor_id": doctor_id,
            "is_available": True,
        })

        if profile is None:
            raise HTTPException(
                status_code=409,
                detail=(
                    "This doctor is not currently marked available. "
                    "Please request fresh recommendations."
                ),
            )

        appointments = mongo_db.carematch_appointments

        # Prevent two bookings for the same doctor at the exact
        # same date and time.
        appointments.create_index(
            [
                ("doctor_id", 1),
                ("appointment_date", 1),
                ("appointment_time", 1),
            ],
            unique=True,
            name="unique_doctor_appointment_slot",
        )

        appointment_doc = {
            "patient_id": patient.id,
            "encounter_id": encounter.id,
            "doctor_id": doctor.id,
            "doctor_name": doctor.full_name,
            "visit_reason": visit_reason,
            "appointment_date": parsed_date.isoformat(),
            "appointment_time": parsed_time,
            "status": "BOOKED",
            "created_by": current_user.id,
            "created_at": datetime.utcnow(),
        }

        saved = appointments.insert_one(appointment_doc)

    except DuplicateKeyError as exc:
        raise HTTPException(
            status_code=409,
            detail=(
                "This doctor already has an appointment at that "
                "exact date and time. Choose another slot."
            ),
        ) from exc

    except PyMongoError as exc:
        raise HTTPException(
            status_code=503,
            detail="Could not save the appointment. Check MongoDB.",
        ) from exc

    return {
        "message": "Appointment saved successfully.",
        "appointment": {
            "id": str(saved.inserted_id),
            "patient_id": patient.id,
            "encounter_id": encounter.id,
            "doctor_id": doctor.id,
            "doctor_name": doctor.full_name,
            "appointment_date": parsed_date.isoformat(),
            "appointment_time": parsed_time,
            "status": "BOOKED",
        },
    }

@router.post("/patients")
def create_hospital_patient(
    request: dict,
    db: Session = Depends(get_db),
    current_user: User = Depends(
    require_patient_registration_permission
),
):
    """
    Register a new patient.

    Automatically creates:
    - Patient master record
    - Initial Encounter
    - Initial Billing record
    - Active QR token

    Patient ID is generated automatically.

    Required:
    - first_name
    - last_name
    - doctor_id

    Optional admission:
    - is_admitted
    - ward_id
    - bed_id
    """

    # -----------------------------------------------------
    # Required fields
    # -----------------------------------------------------
    required_fields = [
        "first_name",
        "last_name",
        "doctor_id",
    ]

    for field in required_fields:
        value = request.get(field)

        if (
            value is None
            or str(value).strip() == ""
        ):
            raise HTTPException(
                status_code=400,
                detail=f"{field} is required.",
            )

    # -----------------------------------------------------
    # Basic values
    # -----------------------------------------------------
    first_name = str(
        request["first_name"]
    ).strip()

    last_name = str(
        request["last_name"]
    ).strip()

    gender = request.get("gender")

    if gender is not None:
        gender = str(gender).strip() or None

    # -----------------------------------------------------
    # Date of birth
    # -----------------------------------------------------
    date_of_birth = parse_date_of_birth(
        request.get("date_of_birth")
    )

    # -----------------------------------------------------
    # Weight
    # -----------------------------------------------------
    weight_kg = parse_weight(
        request.get("weight_kg")
    )

    # -----------------------------------------------------
    # Doctor
    # -----------------------------------------------------
    doctor_id = request.get("doctor_id")

    try:
        doctor_id = int(doctor_id)

    except (TypeError, ValueError):
        raise HTTPException(
            status_code=400,
            detail="doctor_id must be a valid integer.",
        )

    doctor = db.scalar(
        select(User)
        .join(Role, User.role_id == Role.id)
        .where(
            User.id == doctor_id,
            User.is_active.is_(True),
            Role.name == "Doctor",
        )
    )

    if doctor is None:
        raise HTTPException(
            status_code=400,
            detail=(
                "Selected doctor is not an active "
                "registered doctor."
            ),
        )

    # -----------------------------------------------------
    # Admission state
    # -----------------------------------------------------
    is_admitted = parse_boolean(
        request.get("is_admitted"),
        default=False,
    )

    # -----------------------------------------------------
    # Ward / bed
    # -----------------------------------------------------
    ward_id = request.get("ward_id")
    bed_id = request.get("bed_id")

    if ward_id in ("", None):
        ward_id = None

    if bed_id in ("", None):
        bed_id = None

    if is_admitted and ward_id is None:
        raise HTTPException(
            status_code=400,
            detail=(
                "Ward is required when admitting "
                "a patient."
            ),
        )

    if is_admitted and bed_id is None:
        raise HTTPException(
            status_code=400,
            detail=(
                "Bed is required when admitting "
                "a patient."
            ),
        )

    if not is_admitted:
        ward_id = None
        bed_id = None

    # -----------------------------------------------------
    # Validate ward / bed
    # -----------------------------------------------------
    selected_bed = None

    if bed_id is not None:
        try:
            ward_id = (
                int(ward_id)
                if ward_id is not None
                else None
            )

            bed_id = int(bed_id)

        except (TypeError, ValueError):
            raise HTTPException(
                status_code=400,
                detail=(
                    "ward_id and bed_id must be "
                    "valid integers."
                ),
            )

        selected_bed = db.scalar(
            select(Bed).where(
                Bed.id == bed_id,
                Bed.is_active.is_(True),
            )
        )

        if selected_bed is None:
            raise HTTPException(
                status_code=404,
                detail=(
                    "Selected bed not found or inactive."
                ),
            )

        if selected_bed.is_occupied:
            raise HTTPException(
                status_code=409,
                detail="Selected bed is already occupied.",
            )

        if (
            ward_id is not None
            and selected_bed.ward_id != ward_id
        ):
            raise HTTPException(
                status_code=400,
                detail=(
                    "Selected bed does not belong "
                    "to the selected ward."
                ),
            )

    # -----------------------------------------------------

    # -----------------------------------------------------
    patient_code = generate_patient_id(db)

    # -----------------------------------------------------
    qr_token = generate_patient_qr_token()

    while (
        db.scalar(

                Patient.qr_token == qr_token
            )
        )
        is not None
    ):
        qr_token = generate_patient_qr_token()

    now = datetime.utcnow()

    # -----------------------------------------------------
    # Create patient
    # -----------------------------------------------------
    patient = Patient(
        patient_id=patient_code,
        first_name=first_name,
        last_name=last_name,
        date_of_birth=date_of_birth,
        gender=gender,
        weight_kg=weight_kg,
        ward_id=ward_id,
        bed_id=bed_id,
        is_active=True,
        qr_token=qr_token,
        qr_generated_at=now,
        qr_is_active=True,
    )

    try:
        db.add(patient)

        # Flush so patient.id becomes available.

        # -------------------------------------------------

            admitted_at=(
