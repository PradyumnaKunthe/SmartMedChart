def generate_patient_qr_token() -> str:
    """
    Generate a secure opaque QR token for a patient.
    """
    return token_urlsafe(32)


def generate_patient_id(db: Session) -> str:
    """
    Automatically generate a unique SmartMedChart Patient ID.

    Format:
        SMH-PUN-YYMMDD-XXXX
    """
    today = date.today()
    date_code = today.strftime("%y%m%d")
    prefix = f"SMH-PUN-{date_code}-"

    sequence = 1

    while True:
        patient_code = f"{prefix}{sequence:04d}"

        existing_patient = db.scalar(
            select(Patient).where(
                Patient.patient_id == patient_code
            )
        )

        if existing_patient is None:
            return patient_code

        sequence += 1


def parse_date_of_birth(value) -> date | None:
    """
    Convert a date value such as '1979-05-14'
    into a Python date object.
    """
    if value is None or str(value).strip() == "":
        return None

    if isinstance(value, date):
        return value

    try:
        return datetime.strptime(
            str(value).strip(),
            "%Y-%m-%d",
        ).date()

    except ValueError:
        raise HTTPException(
            status_code=400,
            detail=(
                "Invalid date_of_birth. "
                "Use YYYY-MM-DD format."
            ),
        )


def parse_weight(value):
    """
    Convert weight to a numeric value.
    """
    if value is None or str(value).strip() == "":
        return None

    try:
        weight = float(value)

    except (TypeError, ValueError):
        raise HTTPException(
            status_code=400,
            detail="weight_kg must be a valid number.",
        )

    if weight <= 0 or weight > 500:
        raise HTTPException(
            status_code=400,
            detail="weight_kg must be between 0 and 500.",
        )

    return weight


def parse_boolean(value, default=True) -> bool:
    """
    Safely convert incoming boolean-like values.
    """
    if value is None:
        return default

    if isinstance(value, bool):
        return value

    text = str(value).strip().lower()

    if text in {"true", "1", "yes", "y", "active"}:
        return True

    if text in {"false", "0", "no", "n", "inactive"}:
        return False

    raise HTTPException(
        status_code=400,
        detail=f"Invalid boolean value: {value}",
    )


def patient_to_dict(patient: Patient) -> dict:
    """
    Convert a Patient SQLAlchemy object into
    a JSON-safe dictionary.
    """
    return {
        "id": patient.id,
        "patient_id": patient.patient_id,
        "first_name": patient.first_name,
        "last_name": patient.last_name,
        "full_name": (
            f"{patient.first_name} "
            f"{patient.last_name}"
        ).strip(),
        "date_of_birth": patient.date_of_birth,
        "gender": patient.gender,
        "weight_kg": (
            float(patient.weight_kg)
            if patient.weight_kg is not None
            else None
        ),
        "ward_id": patient.ward_id,
        "bed_id": patient.bed_id,
        "is_active": patient.is_active,
        "qr_is_active": patient.qr_is_active,
        "qr_generated_at": patient.qr_generated_at,
    }

@router.get("/dashboard")
def get_hospital_dashboard(
    db: Session = Depends(get_db),
    current_user: User = Depends(
        require_permission("VIEW_PATIENT")
    ),
):
    """
    Return patient statistics for the Hospital Portal.
    """
    total_patient_count = (
        db.scalar(
            select(func.count(Patient.id))
        )
        or 0
    )

    active_patient_count = (
        db.scalar(
            select(
                func.count(Patient.id)
            ).where(
                Patient.is_active.is_(True)
            )
        )
        or 0
    )

    inactive_patient_count = (
        int(total_patient_count)
        - int(active_patient_count)
    )

    return {
        "message": (
            "Hospital dashboard generated successfully."
        ),
        "hospital": {
            "portal": "Hospital Data Portal",
            "status": "ACTIVE",
        },
        "patients": {
            "total": int(total_patient_count),
            "active": int(active_patient_count),
            "inactive": int(inactive_patient_count),
        },
    }

@router.get("/patients")
def get_hospital_patients(
    search: str | None = None,
    active_only: bool = False,
    limit: int = 100,
    db: Session = Depends(get_db),
    current_user: User = Depends(
        require_permission("VIEW_PATIENT")
    ),
):
    """
    Fetch patients from the shared SmartMedChart database.

    Supports searching by:
    - Patient ID
    - First name
    - Last name
    """
    if limit < 1:
        limit = 1

    if limit > 500:
        limit = 500

    query = select(Patient)

    if active_only:
        query = query.where(
            Patient.is_active.is_(True)
        )

    if search:
        search_value = search.strip()

        if search_value:
            pattern = f"%{search_value}%"

            query = query.where(
                or_(
                    Patient.patient_id.ilike(pattern),
                    Patient.first_name.ilike(pattern),
                    Patient.last_name.ilike(pattern),
                )
            )

    query = (
        query
        .order_by(Patient.id.desc())
        .limit(limit)
    )

    patients = db.scalars(query).all()

    return {
        "count": len(patients),
        "patients": [
            patient_to_dict(patient)
            for patient in patients
        ],
    }

@router.get("/patients/{patient_id}")
def get_hospital_patient(
    patient_id: int,
    db: Session = Depends(get_db),
    current_user: User = Depends(
        require_permission("VIEW_PATIENT")
    ),
):
    """
    Return the complete hospital patient profile.

    Includes:
    - Patient information
    - Ward information
    - Bed information
    - Latest encounter
    - Attending doctor
    - Billing
    """

    patient = db.scalar(
        select(Patient).where(
            Patient.id == patient_id
        )
    )

    if patient is None:
        raise HTTPException(
            status_code=404,
            detail="Patient not found.",
        )

    # -----------------------------------------------------
    # Latest encounter
    # -----------------------------------------------------
    encounter = db.scalar(
        select(Encounter)
        .where(
            Encounter.patient_id == patient.id
        )
        .order_by(
            desc(Encounter.created_at)
        )
    )

    # -----------------------------------------------------
    # Doctor
    # -----------------------------------------------------
    doctor = None

    if (
        encounter is not None
        and encounter.doctor_id is not None
    ):
        doctor = db.scalar(
            select(User).where(
                User.id == encounter.doctor_id
            )
        )

    # -----------------------------------------------------
    # Ward
    # -----------------------------------------------------
    ward = None

    if patient.ward_id is not None:
        ward = db.scalar(
            select(Ward).where(
                Ward.id == patient.ward_id
            )
        )

    # -----------------------------------------------------
    # Bed
    # -----------------------------------------------------
    bed = None

    if patient.bed_id is not None:
        bed = db.scalar(
            select(Bed).where(
                Bed.id == patient.bed_id
            )
        )

    # -----------------------------------------------------
    # Billing
    # -----------------------------------------------------
    billing = db.scalar(
        select(PatientBilling)
        .where(
            PatientBilling.patient_id == patient.id
        )
        .order_by(
            desc(PatientBilling.created_at)
        )
    )

    return {
        "patient": {
            "id": patient.id,
            "patient_id": patient.patient_id,

            "first_name": patient.first_name,
            "last_name": patient.last_name,

            "full_name": (
                f"{patient.first_name} "
                f"{patient.last_name}"
            ).strip(),

            "date_of_birth": patient.date_of_birth,
            "gender": patient.gender,

            "weight_kg": (
                float(patient.weight_kg)
                if patient.weight_kg is not None
                else None
            ),

            "ward_id": patient.ward_id,
            "bed_id": patient.bed_id,

            "is_active": patient.is_active,
            "qr_is_active": patient.qr_is_active,
            "qr_generated_at": patient.qr_generated_at,

            "ward": (
                {
                    "id": ward.id,
                    "name": ward.name,
                    "code": ward.code,
                    "description": ward.description,
                    "is_active": ward.is_active,
                }
                if ward is not None
                else None
            ),

            "bed": (
                {
                    "id": bed.id,
                    "bed_number": bed.bed_number,
                    "ward_id": bed.ward_id,
                    "is_active": bed.is_active,
                    "is_occupied": bed.is_occupied,
                }
                if bed is not None
                else None
            ),

            "encounter": (
                {
                    "id": encounter.id,
                    "doctor_id": encounter.doctor_id,
                    "doctor_name": (
                        doctor.full_name
                        if doctor is not None
                        else None
                    ),
                    "encounter_type": encounter.encounter_type,
                    "status": encounter.status,
                    "admitted_at": encounter.admitted_at,
                    "discharged_at": encounter.discharged_at,
                    "discharge_reason": encounter.discharge_reason,
                }
                if encounter is not None
                else None
            ),

            "doctor": (
                {
                    "id": doctor.id,
                    "full_name": doctor.full_name,
                    "username": doctor.username,
                    "email": doctor.email,
                }
                if doctor is not None
                else None
            ),

            "billing": (
                {
                    "id": billing.id,
                    "bill_status": billing.bill_status,
                    "bill_amount": (
                        float(billing.bill_amount)
                        if billing.bill_amount is not None
                        else None
                    ),
                    "paid_at": billing.paid_at,
                    "paid_by": billing.paid_by,
                    "created_at": billing.created_at,
                    "updated_at": billing.updated_at,
                }
                if billing is not None
                else None
            ),
        }
    }

@router.post("/patients")
def create_hospital_patient(
    request: dict,
    db: Session = Depends(get_db),
    current_user: User = Depends(
        require_permission("MANAGE_SYSTEM")
    ),
):
    """
    Register a new patient.

    Automatically creates:
    - Patient master record
    - Initial Encounter
    - Initial Billing record
    - Active QR token

    Patient ID is generated automatically.

    Required:
    - first_name
    - last_name
    - doctor_id

    Optional admission:
    - is_admitted
    - ward_id
    - bed_id
    """

    # -----------------------------------------------------
    # Required fields
    # -----------------------------------------------------
    required_fields = [
        "first_name",
        "last_name",
        "doctor_id",
    ]

    for field in required_fields:
        value = request.get(field)

        if (
            value is None
            or str(value).strip() == ""
        ):
            raise HTTPException(
                status_code=400,
                detail=f"{field} is required.",
            )

    # -----------------------------------------------------
    # Basic values
    # -----------------------------------------------------
    first_name = str(
        request["first_name"]
    ).strip()

    last_name = str(
        request["last_name"]
    ).strip()

    gender = request.get("gender")

    if gender is not None:
        gender = str(gender).strip() or None

    # -----------------------------------------------------
    # Date of birth
    # -----------------------------------------------------
    date_of_birth = parse_date_of_birth(
        request.get("date_of_birth")
    )

    # -----------------------------------------------------
    # Weight
    # -----------------------------------------------------
    weight_kg = parse_weight(
        request.get("weight_kg")
    )

    # -----------------------------------------------------
    # Doctor
    # -----------------------------------------------------
    doctor_id = request.get("doctor_id")

    try:
        doctor_id = int(doctor_id)

    except (TypeError, ValueError):
        raise HTTPException(
            status_code=400,
            detail="doctor_id must be a valid integer.",
        )

    doctor = db.scalar(
        select(User)
        .join(Role, User.role_id == Role.id)
        .where(
            User.id == doctor_id,
            User.is_active.is_(True),
            Role.name == "Doctor",
        )
    )

    if doctor is None:
        raise HTTPException(
            status_code=400,
            detail=(
                "Selected doctor is not an active "
                "registered doctor."
            ),
        )

    # -----------------------------------------------------
    # Admission state
    # -----------------------------------------------------
    is_admitted = parse_boolean(
        request.get("is_admitted"),
        default=False,
    )

    # -----------------------------------------------------
    # Ward / bed
    # -----------------------------------------------------
    ward_id = request.get("ward_id")
    bed_id = request.get("bed_id")

    if ward_id in ("", None):
        ward_id = None

    if bed_id in ("", None):
        bed_id = None

    if is_admitted and ward_id is None:
        raise HTTPException(
            status_code=400,
            detail=(
                "Ward is required when admitting "
                "a patient."
            ),
        )

    if is_admitted and bed_id is None:
        raise HTTPException(
            status_code=400,
            detail=(
                "Bed is required when admitting "
                "a patient."
            ),
        )

    if not is_admitted:
        ward_id = None
        bed_id = None

    # -----------------------------------------------------
    # Validate ward / bed
    # -----------------------------------------------------
    selected_bed = None

    if bed_id is not None:
        try:
            ward_id = (
                int(ward_id)
                if ward_id is not None
                else None
            )

            bed_id = int(bed_id)

        except (TypeError, ValueError):
            raise HTTPException(
                status_code=400,
                detail=(
                    "ward_id and bed_id must be "
                    "valid integers."
                ),
            )

        selected_bed = db.scalar(
            select(Bed).where(
                Bed.id == bed_id,
                Bed.is_active.is_(True),
            )
        )

        if selected_bed is None:
            raise HTTPException(
                status_code=404,
                detail=(
                    "Selected bed not found or inactive."
                ),
            )

        if selected_bed.is_occupied:
            raise HTTPException(
                status_code=409,
                detail="Selected bed is already occupied.",
            )

        if (
            ward_id is not None
            and selected_bed.ward_id != ward_id
        ):
            raise HTTPException(
                status_code=400,
                detail=(
                    "Selected bed does not belong "
                    "to the selected ward."
                ),
            )

    # -----------------------------------------------------
    # Generate Patient ID
    # -----------------------------------------------------
    patient_code = generate_patient_id(db)

    # -----------------------------------------------------
    # Generate unique QR token
    # -----------------------------------------------------
    qr_token = generate_patient_qr_token()

    while (
        db.scalar(
            select(Patient).where(
                Patient.qr_token == qr_token
            )
        )
        is not None
    ):
        qr_token = generate_patient_qr_token()

    now = datetime.utcnow()

    # -----------------------------------------------------
    # Create patient
    # -----------------------------------------------------
    patient = Patient(
        patient_id=patient_code,
        first_name=first_name,
        last_name=last_name,
        date_of_birth=date_of_birth,
        gender=gender,
        weight_kg=weight_kg,
        ward_id=ward_id,
        bed_id=bed_id,
        is_active=True,
        qr_token=qr_token,
        qr_generated_at=now,
        qr_is_active=True,
    )

    try:
        db.add(patient)

        # Flush so patient.id becomes available.
        db.flush()

        # -------------------------------------------------
        # Create initial encounter
        # -------------------------------------------------
        encounter = Encounter(
            patient_id=patient.id,
            doctor_id=doctor_id,
            encounter_type=(
                "INPATIENT"
                if is_admitted
                else "OPD"
            ),
            status="ACTIVE",
            admitted_at=(
                now
                if is_admitted
                else None
            ),
            created_at=now,
            updated_at=now,
        )

        db.add(encounter)

        # -------------------------------------------------
        # Create initial billing record
        # -------------------------------------------------
        billing = PatientBilling(
            patient_id=patient.id,
            bill_status="PENDING",
            bill_amount=None,
            created_at=now,
            updated_at=now,
        )

        db.add(billing)

        # -------------------------------------------------
        # Occupy bed
        # -------------------------------------------------
        if selected_bed is not None:
            selected_bed.is_occupied = True
            selected_bed.updated_at = now

        db.commit()

        db.refresh(patient)
        db.refresh(encounter)
        db.refresh(billing)

    except IntegrityError as exc:
        db.rollback()

        raise HTTPException(
            status_code=409,
            detail=(
                "Patient could not be created because "
                "a unique database value already exists."
            ),
        ) from exc

    return {
        "message": "Patient registered successfully.",

        "patient": {
            "id": patient.id,
            "patient_id": patient.patient_id,
            "first_name": patient.first_name,
            "last_name": patient.last_name,
            "full_name": (
                f"{patient.first_name} "
                f"{patient.last_name}"
            ).strip(),
            "date_of_birth": patient.date_of_birth,
            "gender": patient.gender,
            "weight_kg": (
                float(patient.weight_kg)
                if patient.weight_kg is not None
                else None
            ),
            "ward_id": patient.ward_id,
            "bed_id": patient.bed_id,
            "is_active": patient.is_active,
            "qr_is_active": patient.qr_is_active,
            "qr_generated_at": patient.qr_generated_at,
        },

        "encounter": {
            "id": encounter.id,
            "doctor_id": encounter.doctor_id,
            "doctor_name": (
                encounter.doctor.full_name
                if encounter.doctor is not None
                else doctor.full_name
            ),
            "encounter_type": encounter.encounter_type,
            "status": encounter.status,
            "admitted_at": encounter.admitted_at,
        },

        "billing": {
            "id": billing.id,
            "bill_status": billing.bill_status,
            "bill_amount": (
                float(billing.bill_amount)
                if billing.bill_amount is not None
                else None
            ),
        },
    }

@router.post(
    "/patients/{patient_id}/qr/regenerate"
)
def regenerate_patient_qr(
    patient_id: int,
    db: Session = Depends(get_db),
    current_user: User = Depends(
        require_permission("MANAGE_SYSTEM")
    ),
):
    """
    Generate a new QR token for an active patient.
    """

    patient = db.scalar(
        select(Patient).where(
            Patient.id == patient_id
        )
    )

    if patient is None:
        raise HTTPException(
            status_code=404,
            detail="Patient not found.",
        )

    # A discharged patient must never receive
    # a newly activated QR code.
    if not patient.is_active:
        raise HTTPException(
            status_code=409,
            detail=(
                "Cannot regenerate QR for a "
                "discharged patient."
            ),
        )

    qr_token = generate_patient_qr_token()

    while (
        db.scalar(
            select(Patient).where(
                Patient.qr_token == qr_token
            )
        )
        is not None
    ):
        qr_token = generate_patient_qr_token()

    patient.qr_token = qr_token
    patient.qr_generated_at = datetime.utcnow()
    patient.qr_is_active = True
    patient.updated_at = datetime.utcnow()

    db.commit()
    db.refresh(patient)

    return {
        "message": (
            "Patient QR token regenerated successfully."
        ),
        "patient_id": patient.id,
        "qr_generated_at": patient.qr_generated_at,
        "qr_is_active": patient.qr_is_active,
    }

@router.post("/staff")
def create_staff(
    request: StaffCreateRequest,
    db: Session = Depends(get_db),
    current_user: User = Depends(
        require_permission("MANAGE_USERS")
    ),
):
    require_admin_permission(current_user)

    role_name = request.role.strip().lower()

    if role_name == "nurse":
        actual_role_name = "Nurse"
    elif role_name == "doctor":
        actual_role_name = "Doctor"
    else:
        raise HTTPException(
            status_code=400,
            detail=(
                "Only Nurse and Doctor accounts "
                "can be created here."
            ),
        )

    existing_username = db.scalar(
        select(User).where(
            User.username == request.username.strip()
        )
    )

    if existing_username is not None:
        raise HTTPException(
            status_code=409,
            detail="Username already exists.",
        )

    existing_email = db.scalar(
        select(User).where(
            User.email == request.email.strip()
        )
    )

    if existing_email is not None:
        raise HTTPException(
            status_code=409,
            detail="Email already exists.",
        )

    role = get_role(
        db,
        actual_role_name,
    )

    user = User(
        username=request.username.strip(),
        email=request.email.strip(),
        password_hash=hash_password(
            request.password
        ),
        full_name=request.full_name.strip(),
        role_id=role.id,
        is_active=True,
        created_at=datetime.utcnow(),
        updated_at=datetime.utcnow(),
    )

    db.add(user)

    try:
        db.commit()
        db.refresh(user)

    except IntegrityError as exc:
        db.rollback()

        raise HTTPException(
            status_code=409,
            detail=(
                "Unable to create staff account because "
                "the username or email already exists."
            ),
        ) from exc

    return {
        "message": (
            f"{actual_role_name} account "
            "created successfully."
        ),
        "staff": serialize_staff(user),
    }

@router.post("/assignments")
def create_assignment(
    request: AssignmentCreateRequest,
    db: Session = Depends(get_db),
    current_user: User = Depends(
        require_permission("MANAGE_USERS")
    ),
):
    require_admin_permission(current_user)

    nurse_role = get_role(
        db,
        "Nurse",
    )

    nurse = db.scalar(
        select(User).where(
            User.id == request.nurse_id,
            User.role_id == nurse_role.id,
            User.is_active.is_(True),
        )
    )

    if nurse is None:
        raise HTTPException(
            status_code=404,
            detail="Active nurse not found.",
        )

    patient = db.scalar(
        select(Patient).where(
            Patient.id == request.patient_id,
            Patient.is_active.is_(True),
        )
    )

    if patient is None:
        raise HTTPException(
            status_code=404,
            detail="Active patient not found.",
        )

    existing = db.scalar(
        select(
            NursePatientAssignment
        ).where(
            NursePatientAssignment.nurse_id
            == nurse.id,
            NursePatientAssignment.patient_id
            == patient.id,
        )
    )

    if existing is not None:

        if existing.is_active:
            raise HTTPException(
                status_code=409,
                detail=(
                    "This patient is already "
                    "assigned to this nurse."
                ),
            )

        existing.is_active = True
        existing.assigned_at = datetime.utcnow()

        db.commit()
        db.refresh(existing)

        return {
            "message": (
                "Patient assignment "
                "reactivated successfully."
            ),
            "assignment": {
                "id": existing.id,
                "nurse_id": nurse.id,
                "nurse_name": nurse.full_name,
                "patient_id": patient.id,
                "patient_name": (
                    f"{patient.first_name} "
                    f"{patient.last_name}"
                ).strip(),
                "patient_code": patient.patient_id,
                "is_active": existing.is_active,
            },
        }

    assignment = NursePatientAssignment(
        nurse_id=nurse.id,
        patient_id=patient.id,
        assigned_at=datetime.utcnow(),
        is_active=True,
    )

    db.add(assignment)

    try:
        db.commit()
        db.refresh(assignment)

    except IntegrityError as exc:
        db.rollback()

        raise HTTPException(
            status_code=409,
            detail=(
                "This nurse-patient assignment "
                "already exists."
            ),
        ) from exc

    return {
        "message": (
            "Patient assigned to nurse successfully."
        ),
        "assignment": {
            "id": assignment.id,
            "nurse_id": nurse.id,
            "nurse_name": nurse.full_name,
            "patient_id": patient.id,
            "patient_name": (
                f"{patient.first_name} "
                f"{patient.last_name}"
            ).strip(),
            "patient_code": patient.patient_id,
            "is_active": assignment.is_active,
        },
    }

@router.post("/doctor-assignments")
def assign_patient_to_doctor(
    request: dict,
    db: Session = Depends(get_db),
    current_user: User = Depends(
        require_permission("MANAGE_USERS")
    ),
):
    """
    Assign an active patient to an active doctor.

    The assignment is stored on the patient's ACTIVE
    encounter through Encounter.doctor_id.
    """
    require_admin_permission(current_user)

    doctor_id = request.get("doctor_id")
    patient_id = request.get("patient_id")

    try:
        doctor_id = int(doctor_id)
        patient_id = int(patient_id)
    except (TypeError, ValueError) as exc:
        raise HTTPException(
            status_code=400,
            detail=(
                "doctor_id and patient_id are "
                "required integers."
            ),
        ) from exc

    doctor_role = get_role(
        db,
        "Doctor",
    )

    doctor = db.scalar(
        select(User).where(
            User.id == doctor_id,
            User.role_id == doctor_role.id,
            User.is_active.is_(True),
        )
    )

    if doctor is None:
        raise HTTPException(
            status_code=404,
            detail="Active doctor not found.",
        )

    patient = db.scalar(
        select(Patient).where(
            Patient.id == patient_id,
            Patient.is_active.is_(True),
        )
    )

    if patient is None:
        raise HTTPException(
            status_code=404,
            detail="Active patient not found.",
        )

    now = datetime.utcnow()

    encounter = db.scalar(
        select(Encounter)
        .where(
            Encounter.patient_id == patient.id,
            Encounter.status == "ACTIVE",
        )
        .order_by(
            desc(Encounter.created_at)
        )
    )

    if encounter is None:
        encounter_type = (
            "INPATIENT"
            if patient.bed_id is not None
            else "OPD"
        )

        encounter = Encounter(
            patient_id=patient.id,
            doctor_id=doctor.id,
            encounter_type=encounter_type,
            status="ACTIVE",
            admitted_at=(
                now
                if encounter_type == "INPATIENT"
                else None
            ),
            created_at=now,
            updated_at=now,
        )

        db.add(encounter)

    else:
        encounter.doctor_id = doctor.id
        encounter.updated_at = now

    try:
        db.commit()
        db.refresh(encounter)

    except IntegrityError as exc:
        db.rollback()
        raise HTTPException(
            status_code=409,
            detail=(
                "Patient could not be assigned "
                "to the selected doctor."
            ),
        ) from exc

    return {
        "message": (
            "Patient assigned to doctor successfully."
        ),
        "assignment": {
            "id": encounter.id,
            "doctor_id": doctor.id,
            "doctor_name": doctor.full_name,
            "doctor_username": doctor.username,
            "patient_id": patient.id,
            "patient_code": patient.patient_id,
            "patient_name": (
                f"{patient.first_name} "
                f"{patient.last_name}"
            ).strip(),
            "encounter_type": encounter.encounter_type,
            "encounter_status": encounter.status,
            "assigned_at": encounter.created_at,
        },
    }

@router.get("/workload/nurses")

    db: Session = Depends(get_db),
    current_user: User = Depends(
        require_permission("MANAGE_USERS")
    ),
):
    """Return all active registered nurses with their active patient workload."""

    require_admin_permission(current_user)

    nurse_role = get_role(db, "Nurse")

    nurses = db.scalars(
        select(User)
        .where(
            User.role_id == nurse_role.id,
            User.is_active.is_(True),
        )
        .order_by(User.full_name, User.username)
    ).all()

    result = []

    for nurse in nurses:
        assignments = db.scalars(
            select(NursePatientAssignment)
            .where(
                NursePatientAssignment.nurse_id == nurse.id,
                NursePatientAssignment.is_active.is_(True),
            )
            .order_by(
                desc(NursePatientAssignment.assigned_at)
            )
        ).all()

        patients = []

        for assignment in assignments:
            patient = db.scalar(
                select(Patient).where(
                    Patient.id == assignment.patient_id,
                    Patient.is_active.is_(True),
                )
            )

            if patient is None:
                continue

            patient_data = serialize_workload_patient(patient)
            patient_data["assignment_id"] = assignment.id
            patient_data["assigned_at"] = assignment.assigned_at
            patients.append(patient_data)

        result.append(
            {
                "id": nurse.id,
                "full_name": nurse.full_name,
                "username": nurse.username,
                "email": nurse.email,
                "role": "Nurse",
                "is_active": nurse.is_active,
                "patient_count": len(patients),
                "patients": patients,
            }
        )

    return {
        "count": len(result),
        "nurses": result,
    }

        select(Patient)
        .where(
            Patient.is_active.is_(False)
        )
        .order_by(
            desc(Patient.updated_at)
        )
    ).all()

    def patient_summary(
        patient: Patient,
    ):
        billing = db.scalar(
            select(PatientBilling).where(
                PatientBilling.patient_id
                == patient.id
            )
        )

        active_encounter = db.scalar(
            select(Encounter)
            .where(
                Encounter.patient_id
                == patient.id,
                Encounter.status == "ACTIVE",
            )
            .order_by(
                desc(Encounter.created_at)
            )
        )

        return {
            "id": patient.id,
            "patient_id": patient.patient_id,
            "first_name": patient.first_name,

            ).strip(),

                float(patient.weight_kg)
                if patient.weight_kg is not None
                else None
            ),
            "ward_id": patient.ward_id,
            "bed_id": patient.bed_id,
            "is_active": patient.is_active,
            "qr_is_active": patient.qr_is_active,
            "billing": serialize_billing(
                billing
            ),
            "active_encounter": (
                serialize_encounter(
                    active_encounter
                )
                if active_encounter
                else None
            ),
        }

    return {
        "active": [
            patient_summary(patient)
            for patient in active_patients
        ],

@router.post(
    "/patients/{patient_id}/billing"
)
def create_or_update_patient_bill(
    patient_id: int,
    request: dict,
    db: Session = Depends(get_db),
    current_user: User = Depends(
        require_permission("MANAGE_SYSTEM")
    ),
):
    patient = db.scalar(
        select(Patient).where(
            Patient.id == patient_id
        )
    )

    if patient is None:
        raise HTTPException(
            status_code=404,
            detail="Patient not found.",
        )

    amount = request.get("bill_amount")

    if amount in (None, ""):
        bill_amount = None
    else:
        try:
            bill_amount = float(amount)
        except (TypeError, ValueError):
            raise HTTPException(
                status_code=400,
                detail=(
                    "bill_amount must be "
                    "a valid number."
                ),
            )

        if bill_amount < 0:
            raise HTTPException(
                status_code=400,
                detail=(
                    "bill_amount cannot be negative."
                ),
            )

    billing = db.scalar(
        select(PatientBilling).where(
            PatientBilling.patient_id
            == patient.id
        )
    )

    if billing is None:

        billing = PatientBilling(
            patient_id=patient.id,
            bill_status="PENDING",
            bill_amount=bill_amount,
        )

        db.add(billing)

    else:

        if billing.bill_status == "PAID":
            raise HTTPException(
                status_code=409,
                detail=(
                    "The patient's bill is "
                    "already marked as paid."
                ),
            )

        billing.bill_amount = bill_amount
        billing.bill_status = "PENDING"

    db.commit()
    db.refresh(billing)

    return {
        "message": (
            "Patient bill updated successfully."
        ),
        "billing": serialize_billing(
            billing
        ),
    }

@router.post(
    "/patients/{patient_id}/discharge"
)
def discharge_patient(
    patient_id: int,
    request: dict,
    db: Session = Depends(get_db),
    current_user: User = Depends(
        require_permission("MANAGE_SYSTEM")
    ),
):
    patient = db.scalar(
        select(Patient).where(
            Patient.id == patient_id
        )
    )

    if patient is None:
        raise HTTPException(
            status_code=404,
            detail="Patient not found.",
        )

    if not patient.is_active:
        raise HTTPException(
            status_code=409,
            detail="Patient is already discharged.",
        )

    billing = db.scalar(
        select(PatientBilling).where(
            PatientBilling.patient_id
            == patient.id
        )
    )

    if (
        billing is None
        or billing.bill_status != "PAID"
    ):
        raise HTTPException(
            status_code=409,
            detail=(
                "Patient cannot be discharged "
                "until the bill is marked as paid."
            ),
        )

    discharge_reason = request.get(
        "discharge_reason"
    )

    if discharge_reason is not None:
        discharge_reason = str(
            discharge_reason
        ).strip() or None

    now = datetime.utcnow()

    # -----------------------------------------------------
    # Close active encounters
    # -----------------------------------------------------

    encounters = db.scalars(
        select(Encounter).where(
            Encounter.patient_id == patient.id,
            Encounter.status == "ACTIVE",
        )
    ).all()

    for encounter in encounters:
        encounter.status = "DISCHARGED"
        encounter.discharged_at = now
        encounter.discharge_reason = (
            discharge_reason
        )
        encounter.discharged_by = (
            current_user.id
        )

    # -----------------------------------------------------
    # Close nurse assignments
    # -----------------------------------------------------

    assignments = db.scalars(
        select(NursePatientAssignment).where(
            NursePatientAssignment.patient_id
            == patient.id,
            NursePatientAssignment.is_active.is_(
                True
            ),
        )
    ).all()

    for assignment in assignments:
        assignment.is_active = False

    # -----------------------------------------------------
    # Release bed
    # -----------------------------------------------------

    released_bed_id = patient.bed_id

    if patient.bed_id is not None:

        bed = db.scalar(
            select(Bed).where(
                Bed.id == patient.bed_id
            )
        )

        if bed is not None:
            bed.is_occupied = False
            bed.updated_at = now

    # -----------------------------------------------------
    # Deactivate patient + QR
    # -----------------------------------------------------

    patient.is_active = False
    patient.qr_is_active = False
    patient.updated_at = now

    db.commit()
    db.refresh(patient)

    return {
        "message": (
            "Patient discharged successfully."
        ),
        "patient": {
            "id": patient.id,
            "patient_id": patient.patient_id,
            "full_name": (
                f"{patient.first_name} "
                f"{patient.last_name}"
            ).strip(),
            "is_active": patient.is_active,
            "qr_is_active": patient.qr_is_active,
        },
        "discharge": {
            "discharged_at": now,
            "discharged_by": current_user.id,
            "discharge_reason": (
                discharge_reason
            ),
        },
        "released_bed_id": released_bed_id,
        "closed_assignments": len(
            assignments
        ),
        "closed_encounters": len(
            encounters
        ),
    }

@router.get("/wards")
def get_wards(
    db: Session = Depends(get_db),
    current_user: User = Depends(
        require_permission("MANAGE_SYSTEM")
    ),
):
    """
    Fetch all active wards with bed statistics.
    """

    wards = db.scalars(
        select(Ward)
        .where(
            Ward.is_active.is_(True)
        )
        .order_by(
            Ward.id
        )
    ).all()

    result = []

    for ward in wards:

        total_beds = db.scalar(
            select(
                func.count(Bed.id)
            ).where(
                Bed.ward_id == ward.id,
                Bed.is_active.is_(True),
            )
        ) or 0

        occupied_beds = db.scalar(
            select(
                func.count(Bed.id)
            ).where(
                Bed.ward_id == ward.id,
                Bed.is_active.is_(True),
                Bed.is_occupied.is_(True),
            )
        ) or 0

        available_beds = (
            int(total_beds)
            - int(occupied_beds)
        )

        result.append(
            {
                "id": ward.id,
                "name": ward.name,
                "code": ward.code,
                "description": ward.description,
                "total_beds": int(
                    total_beds
                ),
                "occupied_beds": int(
                    occupied_beds
                ),
                "available_beds": int(
                    available_beds
                ),
            }
        )

    return {
        "count": len(result),
        "wards": result,
    }

        .order_by(

    ).all()

    for bed in beds:

        patient = db.scalar(
            select(Patient).where(
                Patient.bed_id == bed.id,

            "name": ward.name,
            "code": ward.code,
            "description": (
                ward.description
            ),
        },

                len(result) - occupied
            ),

@router.post(
    "/patients/{patient_id}/assign-bed"
)
def assign_patient_bed(
    patient_id: int,
    request: dict,
    db: Session = Depends(get_db),
    current_user: User = Depends(
        require_permission("MANAGE_SYSTEM")
    ),
):
    """
    Assign an active patient to an available bed.

    Updates:
    - Patient ward
    - Patient bed
    - Bed occupied state
    - Active encounter
    """

    patient = db.scalar(
        select(Patient).where(
            Patient.id == patient_id,
            Patient.is_active.is_(True),
        )
    )

    if patient is None:
        raise HTTPException(
            status_code=404,
            detail="Active patient not found.",
        )

    ward_id = request.get("ward_id")
    bed_id = request.get("bed_id")

    try:
        ward_id = int(ward_id)
        bed_id = int(bed_id)
    except (TypeError, ValueError):
        raise HTTPException(
            status_code=400,
            detail=(
                "ward_id and bed_id are "
                "required integers."
            ),
        )

    ward = db.scalar(
        select(Ward).where(
            Ward.id == ward_id,
            Ward.is_active.is_(True),
        )
    )

    if ward is None:
        raise HTTPException(
            status_code=404,
            detail="Ward not found or inactive.",
        )

    bed = db.scalar(
        select(Bed).where(
            Bed.id == bed_id,
            Bed.ward_id == ward.id,
            Bed.is_active.is_(True),
        )
    )

    if bed is None:
        raise HTTPException(
            status_code=404,
            detail=(
                "Bed not found, inactive, or "
                "not part of the selected ward."
            ),
        )

    if bed.is_occupied:

        existing_patient = db.scalar(
            select(Patient).where(
                Patient.bed_id == bed.id,
                Patient.is_active.is_(True),
            )
        )

        if (
            existing_patient is not None
            and existing_patient.id
            != patient.id
        ):
            raise HTTPException(
                status_code=409,
                detail=(
                    "This bed is already assigned "
                    "to another active patient."
                ),
            )

    now = datetime.utcnow()

    previous_bed_id = patient.bed_id

    if (
        previous_bed_id is not None
        and previous_bed_id != bed.id
    ):

        previous_bed = db.scalar(
            select(Bed).where(
                Bed.id == previous_bed_id
            )
        )

        if previous_bed is not None:
            previous_bed.is_occupied = False
            previous_bed.updated_at = now

    patient.ward_id = ward.id
    patient.bed_id = bed.id
    patient.is_active = True
    patient.updated_at = now

    bed.is_occupied = True
    bed.updated_at = now

    encounter = db.scalar(
        select(Encounter)
        .where(
            Encounter.patient_id == patient.id,
            Encounter.status == "ACTIVE",
        )
        .order_by(
            desc(
                Encounter.created_at
            )
        )
    )

    if encounter is None:

        encounter = Encounter(
            patient_id=patient.id,
            encounter_type="INPATIENT",
            status="ACTIVE",
            admitted_at=now,
            created_at=now,
            updated_at=now,
        )

        db.add(encounter)

    else:

        encounter.encounter_type = "INPATIENT"

        if encounter.admitted_at is None:
            encounter.admitted_at = now

        encounter.updated_at = now

    try:

        db.commit()

        db.refresh(patient)
        db.refresh(bed)
        db.refresh(encounter)

    except IntegrityError as exc:

        db.rollback()

        raise HTTPException(
            status_code=409,
            detail=(
                "Bed assignment "
                "could not be completed."
            ),
        ) from exc

    return {
        "message": (
            "Patient assigned to "
            "bed successfully."
        ),
        "patient": {
            "id": patient.id,
            "patient_id": patient.patient_id,
            "full_name": (
                f"{patient.first_name} "
                f"{patient.last_name}"
            ).strip(),
            "ward_id": patient.ward_id,
            "bed_id": patient.bed_id,
            "is_active": patient.is_active,
        },
        "ward": {
            "id": ward.id,
            "name": ward.name,
            "code": ward.code,
        },
        "bed": {
            "id": bed.id,
            "bed_number": bed.bed_number,
            "is_occupied": bed.is_occupied,
        },
        "encounter": {
            "id": encounter.id,
            "encounter_type": (
                encounter.encounter_type
            ),
            "status": encounter.status,
            "admitted_at": encounter.admitted_at,
        },
        "previous_bed_id": previous_bed_id,
    }


@router.post(

)
def release_patient_bed(
    patient_id: int,
    db: Session = Depends(get_db),
    current_user: User = Depends(

    ),

    """

    patient = db.scalar(
        select(Patient).where(
            Patient.id == patient_id,
            Patient.is_active.is_(True),
        )
    )

    if patient is None:
        raise HTTPException(
            status_code=404,
            detail="Active patient not found.",
        )

    if patient.bed_id is None:
        return {
            "message": (
                "Patient has no assigned bed."
            ),
            "patient_id": patient.id,
            "bed_id": None,
        }

    bed = db.scalar(
        select(Bed).where(
            Bed.id == patient.bed_id
        )
    )

    previous_bed_id = patient.bed_id

    if bed is not None:
        bed.is_occupied = False
        bed.updated_at = datetime.utcnow()

    patient.bed_id = None
    patient.ward_id = None
    patient.updated_at = datetime.utcnow()

    db.commit()

    return {
        "message": (
            "Patient bed released successfully."
        ),
        "patient_id": patient.id,
        "released_bed_id": previous_bed_id,
    }
